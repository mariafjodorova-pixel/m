    const PLANS = {
        RA1: {
            low: "Medica Solo Bronze A",
            bcbs: "Blue Plus Southeast MN Bronze HSA"
        },
        RA3: {
            low: "Medica Solo Bronze A",
            bcbs: "Blue Plus Southeast MN Bronze HSA"
        },
        RA8: {
            low: "Blue Plus Metro MN Bronze HSA"
        }
    };
    const fmt = n => "$" + n.toLocaleString("en-US", {
        minimumFractionDigits: 2,
        maximumFractionDigits: 2
    });
    const net = (p, a) => Math.max(0, p - a);
    const areaSel = document.getElementById("area"),
        ageSel = document.getElementById("age");
    DATA.RA3.forEach(r => {
        const o = document.createElement("option");
        o.value = r.age;
        o.textContent = r.age;
        ageSel.appendChild(o);
    });
    ageSel.value = "40";

    function planCard(name, tag, prem, allow) {
        return `<div class="plan"><h3>${name} <span class="tag">${tag}</span></h3>
  <div class="math"><div><small>Plan price</small><strong>${fmt(prem)}</strong></div>
  <div><small>Shattuck pays</small><strong>${fmt(Math.min(allow,prem))}</strong></div>
  <div class="you"><small>You pay</small><strong>${fmt(net(prem,allow))}</strong></div></div></div>`;
    }

    function renderCalc() {
        const ra = areaSel.value,
            row = DATA[ra].find(r => r.age === ageSel.value),
            p = PLANS[ra];
        let html = planCard(p.low, ra === "RA8" ? "Lowest-cost Bronze · Blue Cross" : "Lowest-cost Bronze", row.low, row.allow);
        if (row.bcbs) html += planCard(p.bcbs, "Blue Cross Bronze with HSA", row.bcbs, row.allow);
        document.getElementById("results").innerHTML = html;
    }

    function renderTable(ra) {
        const p = PLANS[ra],
            two = !!p.bcbs,
            sel = ageSel.value;
        document.getElementById("tblnote").textContent = two ?
            `Allowance is the same for both plans. Plans shown: ${p.low} (lowest cost) and ${p.bcbs}.` :
            `Plan shown: ${p.low}, the lowest-cost Bronze plan in this area.`;
        let h = `<thead><tr><th>Age</th><th>Shattuck allowance</th><th>${two?"Lowest-cost Bronze":"Bronze plan price"}</th><th>You pay</th>${two?"<th>Blue Cross Bronze HSA</th><th>You pay</th>":""}</tr></thead><tbody>`;
        DATA[ra].forEach(r => {
            if (typeof r.allow === "string") { // ratio check row, not a price row
                h += `<tr><td>${r.age}</td><td colspan="${two?5:3}" style="text-align:left">${r.allow}</td></tr>`;
                return;
            }
            h += `<tr class="${r.age===sel?"hl":""}"><td>${r.age}</td><td>${fmt(r.allow)}</td><td>${fmt(r.low)}</td><td class="you">${fmt(net(r.low,r.allow))}</td>${two?`<td>${fmt(r.bcbs)}</td><td class="you">${fmt(net(r.bcbs,r.allow))}</td>`:""}</tr>`;
        });
        document.getElementById("tbl").innerHTML = h + "</tbody>";
    }
    let tabRA = "RA3";

    function selectTab(ra) {
        tabRA = ra;
        document.querySelectorAll("#tabs button").forEach(b => b.setAttribute("aria-selected", b.dataset.ra === ra ? "true" : "false"));
        renderTable(ra);
    }
    document.getElementById("tabs").addEventListener("click", e => {
        const b = e.target.closest("button");
        if (b) selectTab(b.dataset.ra);
    });
    areaSel.addEventListener("change", () => {
        renderCalc();
        selectTab(areaSel.value);
    });
    ageSel.addEventListener("change", () => {
        renderCalc();
        renderTable(tabRA);
    });
    renderCalc();
    selectTab("RA3");
</script>
